# SCPT-RL Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Train a Sparse Coupled Placement Transformer (PPO-EAL) to place PCB components on a grid from KiCad board files, subject to hard DRC (action-masked) and soft Lagrangian constraints.

**Architecture:** Rust `pcb_parser` wheel exposes a canonical IR + geometry primitives via PyO3. Python `env/` wraps it as a Gymnasium env with numpy-hard DRC masks. `model/` is a heterogeneous PyG encoder feeding a cross-attention placement policy over legal grid cells. `agent/` runs PPO-EAL with a momentum dual updater. Training cycles boards sequentially; `pcb_router` wheel evaluates placements post-training.

**Tech Stack:** Rust (nom, PyO3, maturin) · Python 3.11 · PyTorch 2.x · torch-geometric 2.4+ · Gymnasium · numpy · PyYAML · wandb · matplotlib · pytest

## Global Constraints

- Python ≥ 3.10 (CI runs 3.11)
- `maturin develop --release` for both Rust wheels; never declared as deps in main `pyproject.toml`
- Sign convention: `clipped_ppo_surrogate` returns maximize-convention value; `total_loss` applies negation **once**. Guarded by dedicated test.
- Action mask is stored per step in rollout buffer; never recomputed during loss pass.
- BC and RL use the same area-descending, cluster-contiguous placement order.
- All-illegal action mask → required handler (not optional): log + penalty + terminate.
- Rust bridge (`src/scpt/rust_bridge/`) re-exports only — no logic, no caching, no transformation.
- Native PyO3 attribute access is the hot path; `to_json()` is opt-in debug only.
- Any Tier 2 term whose v1 definition depends only on a DoF the action space doesn't expose gets weight 0 until that DoF ships. `orient_score` escapes via its position-sensitive definition.
- `router.enabled` must be set in config; router is eval-only in v1.

```rust
#[pyfunction]
pub fn load_kicad_pcb(path: &str) -> PyResult<ir::PcbDesign> {
    let src = std::fs::read_to_string(path)
        .map_err(|e| pyo3::exceptions::PyIOError::new_err(e.to_string()))?;
    let sexprs = kicad::lexer::tokenize(&src)
        .map_err(|e| pyo3::exceptions::PyValueError::new_err(format!("{e:?}")))?;
    kicad::parse_design(&sexprs)
        .map_err(|e| pyo3::exceptions::PyValueError::new_err(format!("{e:?}")))
}
```

```rust
pub fn hpwl(d: &PcbDesign) -> f64 {
    d.nets.iter().map(|net| {
        let world_pads: Vec<Vec2> = net.pads.iter().filter_map(|pr| {
            d.placement.positions[pr.0].as_ref().map(|pl| {
                let pad = &d.components[pr.0].footprint.pads[pr.1];
```

```python
import numpy as np

def compute_action_mask(design, active_comp_idx, grid_resolution, min_spacing, placed_mask):
    H, W = placed_mask.shape
    mask = np.ones((H, W), dtype=np.float32)

    # (a)+(b) precomputed at board level — for v1, board is Rect, no keepouts
    # (extend via rasterize polygons into a static mask cached at reset)
    bounds = design.board.bounds
    # outside-board cells already covered by caller constructing placed_mask to board grid

    # (c) courtyard overlap with placed components
    active = design.components[active_comp_idx]
    cy_w = active.footprint.courtyard.width + min_spacing
    cy_h = active.footprint.courtyard.height + min_spacing

    for comp_idx, comp in enumerate(design.components):
        if design.placement.positions[comp_idx] is None:
            continue
        pl = design.placement.positions[comp_idx]
        cx_cells = int((pl.center.x - bounds.x) / grid_resolution)
        cy_cells = int((pl.center.y - bounds.y) / grid_resolution)
        half_w = int((comp.footprint.courtyard.width + min_spacing) / (2 * grid_resolution))
        half_h = int((comp.footprint.courtyard.height + min_spacing) / (2 * grid_resolution))
        # active comp's exclusion zone at each candidate cell must not overlap this placed AABB
        r_lo_x = max(0, cx_cells - half_w - int(cy_w / (2 * grid_resolution)))
        r_hi_x = min(W, cx_cells + half_w + int(cy_w / (2 * grid_resolution)) + 1)
        r_lo_y = max(0, cy_cells - half_h - int(cy_h / (2 * grid_resolution)))
        r_hi_y = min(H, cy_cells + half_h + int(cy_h / (2 * grid_resolution)) + 1)
        mask[r_lo_y:r_hi_y, r_lo_x:r_hi_x] = 0.0

    # (d) occupied
    mask[placed_mask] = 0.0
    return mask
```

Use numpy broadcasting (vectorize the placed-component loop by stacking AABB expansions).

```python
import pcb_parser

class ConstraintOrchestrator:
    def __init__(self, design, cfg):
        self.design = design
        self.cfg = cfg
        self.partition_spec = cfg.partition_spec
        self.min_spacing = cfg.min_spacing_mm
        self.sym_pairs = cfg.sym_pairs
        self.w_orient = cfg.w_orient
        self.w_decap = cfg.w_decap
        self.w_therm = cfg.w_therm
        self.w_sym = cfg.w_sym
        self.decap_radius = cfg.decap_radius_mm

    def step_costs(self, active_comp_idx, x, y, rot):
        # Place active component tentatively
        pl = Placement(center=Vec2(x, y), rotation_deg=rot, layer=Layer.F_Cu)
        self.design.placement.positions[active_comp_idx] = pl
        try:
            ref = self.design.components[active_comp_idx].ref_des
            costs = {
                "c_clearance": pcb_parser.clearance_cost(self.design, self.min_spacing),
                "c_partition": pcb_parser.partition_cut_cost(self.design, self.partition_spec),
                "c_hpwl":      pcb_parser.hpwl_incremental(self.design, ref),
                "r_tier2": (
                    self.w_orient * pcb_parser.orient_score(self.design)
                    + self.w_decap  * pcb_parser.decap_proximity_score(self.design, self.decap_radius)
                    + self.w_therm  * pcb_parser.thermal_score(self.design)
                    + self.w_sym    * pcb_parser.symmetry_score(self.design, self.sym_pairs)
                ),
            }
            if hasattr(self.partition_spec, "budget"):
                costs["b_partition"] = self.partition_spec.budget
            return costs
        finally:
            # leave placement in place (env tracks it); caller can reset
            pass
```

```python
import numpy as np
import torch
from .action_mask import compute_action_mask
from .constraints import ConstraintOrchestrator
from scpt.rust_bridge.parser_client import load_kicad_pcb

class PcbPlacementEnv(gym.Env):
    metadata = {"render_modes": []}

    def __init__(self, board_path, cfg):
        super().__init__()
        self.cfg = cfg
        self.design = load_kicad_pcb(board_path)
        self.grid_res = cfg.grid_resolution_mm
        b = self.design.board.bounds
        self.H = int(b.h / self.grid_res)
        self.W = int(b.w / self.grid_res)
        self.action_space = gym.spaces.Discrete(self.H * self.W)

        self.orchestrator = ConstraintOrchestrator(self.design, cfg)
        self.env_state = EnvState(placed_count=0, step_idx=0)
        self._placement_order = list(self.design.placement.placement_order)

    def reset(self, *, seed=None, options=None):
        super().reset(seed=seed)
        self.env_state = EnvState(placed_count=0, step_idx=0)
        for i in range(len(self.design.placement.positions)):
            self.design.placement.positions[i] = None
        return self._build_obs(), {}

    def step(self, action: int):
        cell_y = action // self.W
        cell_x = action % self.W
        x = (cell_x + 0.5) * self.grid_res + self.design.board.bounds.x
        y = (cell_y + 0.5) * self.grid_res + self.design.board.bounds.bounds.y
        active_idx = self._placement_order[self.env_state.step_idx]
        costs = self.orchestrator.step_costs(active_idx, x, y, rot=0.0)
        self.env_state.placed_count += 1
        self.env_state.step_idx += 1
        terminated = self.env_state.placed_count == len(self._placement_order)
        reward = costs.get("r_tier2", 0.0) - sum(costs[k] for k in costs if k.startswith("c_"))
        return self._build_obs(), reward, terminated, False, {"costs": costs}

    def _build_obs(self):
        placed_mask = np.zeros((self.H, self.W), dtype=bool)
        # fill from design.placement.positions
        for idx, pl in enumerate(self.design.placement.positions):
            if pl is not None:
                cx = int((pl.center.x - self.design.board.bounds.x) / self.grid_res)
                cy = int((pl.center.y - self.design.board.bounds.y) / self.grid_res)
                if 0 <= cx < self.W and 0 <= cy < self.H:

        active_idx = self._placement_order[self.env_state.step_idx] if \
```

```python
def clipped_ppo_surrogate(ratio, advantages, clip_eps):
    """Maximize-convention. Returns scalar mean."""
    unclipped = ratio * advantages
    clipped = torch.clamp(ratio, 1 - clip_eps, 1 + clip_eps) * advantages
    return torch.minimum(unclipped, clipped).mean()

def constraint_surrogate(j_c_pi_k, constraint_advantages, budget, gamma):
    """phi_c_i = j_c_pi_k + (1/(1-gamma)) * E[A_c] - b_i.
    Advantages already masked to legal actions by caller."""
    return j_c_pi_k + (1.0 / (1.0 - gamma)) * constraint_advantages - budget

def augmented_lagrangian_penalty(phi_c, lam, sigma):
    inner = torch.clamp(lam / sigma + phi_c, min=0.0)
    return (sigma / 2.0) * (inner.pow(2) - (lam / sigma).pow(2))

def total_loss(reward_surrogate, constraint_phis, lambdas, sigma):
    """Minimize-convention. Negation applied exactly once here."""
    penalty = sum(augmented_lagrangian_penalty(constraint_phis[k], lambdas[k], sigma)
                  for k in constraint_phis)
    return -reward_surrogate + penalty
```

```python
from dataclasses import dataclass, field

@dataclass
class MomentumDualUpdater:
    constraint_names: list
    alpha: float
    ema_decay: float
    _lambdas: dict = field(default_factory=dict)
    _ema: dict = field(default_factory=dict)

    def __post_init__(self):
        for n in self.constraint_names:
            self._lambdas[n] = 0.0
            self._ema[n] = 0.0

    def update(self, phi_c_by_name: dict) -> dict:
        out = {}
        for n in self.constraint_names:
            phi = phi_c_by_name.get(n, 0.0)
            self._ema[n] = self.ema_decay * self._ema[n] + (1 - self.ema_decay) * phi
            self._lambdas[n] = max(0.0, self._lambdas[n] + self.alpha * self._ema[n])
            out[n] = self._lambdas[n]
        return out
```
